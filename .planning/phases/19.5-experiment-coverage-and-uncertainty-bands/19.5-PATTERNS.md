```python
def run_seed_band(
    runner: Callable[[int], pd.DataFrame], seeds: Sequence[int]
) -> pd.DataFrame:
    """Run `runner` once per seed and concatenate the results into one band."""
    frames: list[pd.DataFrame] = []
    for seed in seeds:
        frame = runner(seed).copy()
        frame["seed"] = seed
        frames.append(frame)
    return pd.concat(frames, ignore_index=True)
```
`run_seed_band`'s own docstring (`_io.py:196-211`) **explicitly names E4/E6 as the deferred

`--seeds` + `--check` together. **No change to `run_seed_band` or `parse_seed_list` is needed or
wanted** — this is the entire point of them being the shared primitive.

```python
def _build_computed_cpr_row(config_key, n_cameras, n_frames, normal_fixed,
                             refine_intrinsics, shared_interface) -> dict:
    """Never hand-rolls a grouping and never uses scipy's generic FD colorer —
    n_params/n_groups read straight off the library's own functions."""
    detections = _make_detections(n_cameras, n_frames, visibility=1.0, seed=0)
    camera_order = [f"cam{i}" for i in range(n_cameras)]
    frame_order = list(range(n_frames))
    jac_sparsity = build_jacobian_sparsity(
        detections, reference_camera="cam0", camera_order=camera_order,
        frame_order=frame_order, min_corners=1,
        refine_intrinsics=refine_intrinsics, normal_fixed=normal_fixed,
        shared_interface=shared_interface,
    )
    groups = build_structural_column_groups(
        jac_sparsity, n_cameras, n_frames,
        refine_intrinsics=refine_intrinsics, normal_fixed=normal_fixed,
        shared_interface=shared_interface,
    )
    n_params = int(jac_sparsity.shape[1])
    n_groups = int(groups.max()) + 1
    return {..., "n_params": n_params, "n_groups": n_groups,
            "fd_reduction": n_params / n_groups, "record_source": "computed"}
```
